/**
 * BUSINESS HUB – one script for many businesses. Each business logs in with email + a code
 * sent by email, and gets its own Google Sheet in the folder "Business accounts".
 * Setup (once):
 *  1. Create a blank Google Sheet named "Business Hub". Extensions → Apps Script → paste this file. Save.
 *  2. Choose  setupHub  in the function list → ▶ Run → allow permissions.
 *  3. Deploy → New deployment → Web app, Execute as: Me, Who has access: Anyone → copy the /exec link.
 * Invite someone: in the Sheet, menu  Business Hub → Create invite link  → send the link on WhatsApp.
 * Turn someone off: Accounts tab → Active = no.
 *
 * Original notes – My Business – Estimate Maker: Google Sheets storage (v4: customer page + staff attendance)
 * All app data lives in this spreadsheet:
 *   Estimates  – one row per estimate/invoice (full data in the last column)
 *   Items      – every line item of every estimate (for your own analysis)
 *   Payments   – every payment received
 *   Clients    – client list (name, phone, address, GSTIN)
 *   Catalog    – your item catalog. You can edit this tab directly in Sheets.
 *   Packages   – saved packages
 *   Photos     – catalog photos and seal/signature image
 *   Settings   – business details, layout settings, number counters
 *
 * Setup / update:
 * 1. Extensions → Apps Script → paste this file (replace everything). Set SECRET.
 * 2. Deploy → Manage deployments → ✏️ Edit → Version: New version → Deploy
 *    (first time: Deploy → New deployment → Web app, Execute as Me, Access Anyone).
 */

/* ---------- key/value rows split over several cells ---------- */
function kvRead_(sheet) {
  const v = sheet.getDataRange().getValues(), o = {};
  for (let i = 1; i < v.length; i++) if (v[i][0] !== '') o[v[i][0]] = { at: +v[i][1] || 0, raw: v[i].slice(2).join('') };
  return o;
}
function kvWrite_(sheet, key, str) {
  const v = sheet.getRange(1, 1, sheet.getLastRow(), 1).getValues().map(r => r[0]);
  let i = v.indexOf(key), row;
  if (i > 0) { row = i + 1; const lc = sheet.getLastColumn(); if (lc > 2) sheet.getRange(row, 3, 1, lc - 2).clearContent(); }
  else { row = sheet.getLastRow() + 1; }
  const c = chunks_(str), now = Date.now();
  sheet.getRange(row, 1, 1, 2 + c.length).setValues([[key, now].concat(c)]);
  return now;
}
function kvDelete_(sheet, key) {
  const v = sheet.getRange(1, 1, sheet.getLastRow(), 1).getValues().map(r => r[0]);
  const i = v.indexOf(key); if (i > 0) sheet.deleteRow(i + 1);
}

function doPost(e) {
  const b = JSON.parse(e.postData.contents);
  if (b.action === 'code') return out_(sendCode_(b));
  if (b.action === 'verify') return out_(verify_(b));
  if (b.action === 'logout') { if (b.key) HP_().deleteProperty('ses:' + b.key); return out_({ ok: true }); }
  if (!auth_(String(b.key || ''))) return out_({ ok: false, error: 'Please log in again' });
  if (b.action === 'me') { const a = accRow_(6, CU_); return out_({ ok: true, biz: a.r[0], email: a.r[1] }); }
  if (b.action === 'ai') return out_(aiCall_(b));
  ensureV2_();
  const lock = LockService.getScriptLock(); lock.waitLock(25000);
  try {
    if (b.action === 'save' || b.action === 'delete') {
      const s = est_(), ids = s.getRange(1, 1, s.getLastRow(), 1).getValues().map(r => r[0]);
      const d = b.data || {}, i = ids.indexOf(b.id);
      if (b.action === 'save' && i > 0) { try { mergeRemote_(d, JSON.parse(s.getRange(i + 1, 16).getValue() || '{}')); } catch (x) { } }
      const paid = (+d.adv || 0) + (d.pays || []).reduce((a, p) => a + (+p.amt || 0), 0);

        ? [b.id, '', '', '', '', '', '', '', '', '', '', '', '', Date.now(), 'yes', '']
        : [b.id, d.no || '', d.doctype || '', d.date || '', d.client || '', d.phone || '', d.ename || d.etype || '', d.edate || '', d.venue || '', d.status || 'Draft', +d.total || 0, paid, (+d.total || 0) - paid, d.saved || Date.now(), '', JSON.stringify(d)];
      if (i > 0) s.getRange(i + 1, 1, 1, row.length).setValues([row]); else s.appendRow(row);
      writeDerived_(b.id, b.action === 'delete' ? null : d);
      try { calSync_(b.id, b.action === 'delete' ? null : d); } catch (x) { }
    } else if (b.action === 'calsyncall') {
      return out_({ ok: true, count: calSyncAll_() });
    } else if (b.action === 'setting') {
      const n = b.name;
      if (n === 'catalog') catWrite_(b.value || []);
      else if (n === 'packages') pkgWrite_(b.value || []);
      else if (n === 'stamp' || n.indexOf('ph:') === 0) {
        const key = n === 'stamp' ? '__stamp' : n.slice(3);
        if (b.value) kvWrite_(ph_(), key, b.value); else kvDelete_(ph_(), key);
      } else if (n === 'seq' || n === 'iseq') {
        const kv = kvRead_(set_()), cur = kv[n] ? +JSON.parse(kv[n].raw) : 0;
        if (+b.value > cur) kvWrite_(set_(), n, JSON.stringify(+b.value));
      } else kvWrite_(set_(), n, JSON.stringify(b.value));
    }
    return out_({ ok: true });
  } finally { lock.releaseLock(); }
}

// Runs every 15 minutes: sends a notification for reminders that are due (supports repeating reminders).
function remOn_(r, td) {
  if (!r.d || r.d > td) return false;
  if (!r.rep) return true;
  if (r.rep === 'd') return true;
  const a = new Date(r.d + 'T00:00:00'), b = new Date(td + 'T00:00:00');
  if (r.rep === 'w') return Math.round((b - a) / 864e5) % 7 === 0;
  if (r.rep === 'm') return a.getDate() === b.getDate();
  return false;
}
function remindPush() {
  const tz = 'Asia/Kolkata', now = new Date(), td = Utilities.formatDate(now, tz, 'yyyy-MM-dd'), hm = Utilities.formatDate(now, tz, 'HH:mm');
  const P = PropertiesService.getScriptProperties(), sent = JSON.parse(P.getProperty('remSent') || '{}');
  plan_().rem.forEach(r => {
    if (r.done || !remOn_(r, td)) return;
    const day = r.rep ? td : r.d, key = r.id + '|' + day + '|' + (r.tm || '');
    if (sent[key]) return;
    if (day < td) { sent[key] = 1; return; }          // missed old one-off: don't flood
    if ((r.tm || '08:00') > hm) return;              // not yet time
    push_('⏰ ' + r.t, (r.note ? r.note + ' · ' : '') + (r.tm || 'Today'));
    sent[key] = 1;
  });
  const keep = {}; Object.keys(sent).slice(-400).forEach(k => keep[k] = 1);
  P.setProperty('remSent', JSON.stringify(keep));
}

/* ---------- AI assistant (Gemini) ---------- */
function aiCall_(b) {
  if (!GEMINI_KEY) return { ok: false, error: 'AI key not set in Apps Script (GEMINI_KEY)' };
  const P = PropertiesService.getScriptProperties(), good = P.getProperty('aiModel');
  const list = good ? [good].concat(GEMINI_MODELS.filter(m => m !== good)) : GEMINI_MODELS;
  const parts = [{ text: String(b.prompt || '').slice(0, 60000) }];
  (b.images || (b.image ? [b.image] : [])).slice(0, 4).forEach(im => { const m = String(im).match(/^data:(image\/[a-z+]+);base64,(.+)$/); if (m) parts.push({ inline_data: { mime_type: m[1], data: m[2] } }); });
  const body = { contents: [{ role: 'user', parts }], generationConfig: { temperature: b.json ? 0.2 : 0.6, maxOutputTokens: 8192 } };
  if (b.system) body.system_instruction = { parts: [{ text: String(b.system).slice(0, 20000) }] };
  if (b.json) body.generationConfig.responseMimeType = 'application/json';
  let last = '';
  for (const m of list) {
    try {
      const r = UrlFetchApp.fetch('https://generativelanguage.googleapis.com/v1beta/models/' + m + ':generateContent', { method: 'post', contentType: 'application/json', muteHttpExceptions: true, headers: { 'x-goog-api-key': GEMINI_KEY }, payload: JSON.stringify(body) });
      const code = r.getResponseCode(), j = JSON.parse(r.getContentText() || '{}');
      if (code === 200) { const t = (((j.candidates || [])[0] || {}).content || {}).parts || []; P.setProperty('aiModel', m); return { ok: true, text: t.map(x => x.text || '').join(''), model: m }; }
      last = (j.error && j.error.message) || ('HTTP ' + code);
      if (code === 429) return { ok: false, error: 'AI daily free limit reached – try again later' };
    } catch (x) { last = String(x); }
  }
  return { ok: false, error: 'AI not available: ' + last };
}
